Example: create checkout

# Create a checkout session
curl https://api.paymo.africa/v1/checkout/sessions \
  -H "Authorization: Bearer sk_test_...9f2K" \
  -d amount=2860 \
  -d currency=KES \
  -d channels=mpesa,card,wallet \
  -d reference=KC-8842
const paymo = require('paymo')('sk_test_...9f2K');
const session = await paymo.checkout.sessions.create({
  amount: 2860,

  channels: ['mpesa', 'card'],
  reference: 'KC-8842'
});
$paymo = new \Paymo\Client('sk_test_...9f2K');
$session = $paymo->checkout->create([
  'amount' => 2860,
  'currency' => 'KES',
  'channels' => ['mpesa','card']
]);
